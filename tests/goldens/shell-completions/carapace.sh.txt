#!/bin/sh

set -e
fish -c 'set -x fish_update_completions_detach true; fish_update_completions; set -u fish_update_completions_detach'
{
    echo
    env CARAPACE_EXCLUDES=chezmoi carapace _carapace bash </dev/null || true
    echo

} >'C:\Users\Lev\.bash_completion'

{
    echo
    env CARAPACE_EXCLUDES=chezmoi carapace _carapace fish </dev/null || true
    echo

} >'C:\Users\Lev\.cache\fish\completions\_generated.fish'

{
    echo
    env CARAPACE_EXCLUDES=chezmoi carapace _carapace zsh </dev/null || true
    echo

} >'C:\Users\Lev\.oh-my-zsh\completions\_generated'

{
    echo
    env CARAPACE_EXCLUDES=chezmoi carapace _carapace nushell </dev/null || true
    echo

} >'C:\Users\Lev\.config\nushell\autoload\_generated.nu'
